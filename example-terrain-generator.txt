# Example hook in hole_manager.gd or player.gd
func setup_terrain(blueprint_data: Dictionary) -> void:
	var tee = Vector3(blueprint_data.holes[0].tee_position)
	var pin = Vector3(blueprint_data.holes[0].pin_position)
	var waypoints = blueprint_data.holes[0].assets.rough_zones.map(
		func(r): return Vector3(r.centerline_start[0], r.centerline_start[1], r.centerline_start[2])
	)
	# Fallback to path if rough_zones not yet populated
	if waypoints.is_empty():
		waypoints = [tee, pin] 
		
	var terrain = $TerrainGenerator as TerrainGenerator
	terrain.build_from_waypoints(waypoints, pin, tee)
